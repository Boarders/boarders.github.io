Kripke.Examples
import Kripke.Frame


universe u


namespace Kripke


abbrev frameOfPreorder (W : Type u) [Preorder W]
    (v    : W → ℕ → Prop)
    (mono : ∀ {w w' : W} {i : ℕ}, w ≤ w' → v w i → v w' i) : Frame W where
  val    := v
  le_val := mono


/-!
## Example: Topological Kripke Frame

For any topological space `X` and family of opens `p : ℕ → Opens X`, the
dual of the opens lattice `(Opens X)ᵒᵈ` — ordered by **reverse** inclusion
(`U ≤ V` iff `V ⊆ U`, i.e. `V` is a restriction of `U`) — is a
`Kripke.Frame` via `frameOfPreorder`.

The valuation `val U i = (U ⊆ p i)` says "variable `i` holds on `U`" means
the entire open set `U` lies inside the extent `p i` of variable `i`.
Monotonicity holds: if `V ⊆ U ⊆ p i` then `V ⊆ p i`.
-/


open TopologicalSpace


/-- Kripke frame on `(Opens X)ᵒᵈ` (reverse-inclusion order).
    Variable `i` holds on `U` iff `U ⊆ p i`. -/
abbrev opensFrame (X : Type u) [TopologicalSpace X]
    (p : ℕ → Opens X) : Frame (Opens X)ᵒᵈ :=
  frameOfPreorder _
    (fun U i => OrderDual.ofDual U ≤ p i)
    (fun h hv => (OrderDual.ofDual_le_ofDual.mpr h).trans hv)


end Kripke
